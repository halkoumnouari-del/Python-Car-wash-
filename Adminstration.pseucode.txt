Adminstration pseucode :

Start service menu 
 While true do :
  Print "Welcome to Crystal Car Adminstartion"
  Print "1.veiw services"
  Print "2.add service"
  Print "3.Modifiy service"
  Print "4.Delet service"
  Print "5.Back to the main menu"
  Prompt "Hi service admin what do you want" into service_update 

  IF service_update EQUALS 1 THEN
            OPEN "services.txt" FOR READING
            FOR EACH line IN file
                OUTPUT line
            ENDFOR
            CLOSE file
          
  ELSE IF service_update EQUALS 2 THEN
            OUTPUT "Enter service ID: "
            INPUT s_id
            OUTPUT "Enter service name: "
            INPUT s_name
            OUTPUT "Enter service price: "
            INPUT s_price

            CREATE service_record FROM s_id, s_name, s_price
            CALL append_regestrations WITH "services.txt" AND service_record
            OUTPUT "Service added successfully"

  ELSE IF service_update EQUALS 3 THEN
            OUTPUT "Enter service ID to modify: "
            INPUT target_id
            SET found TO False
            INITIALIZE updated_list AS EMPTY LIST

            OPEN "services.txt" FOR READING
            FOR EACH line IN file
                SPLIT line BY "|" INTO parts
                IF parts[0] EQUALS target_id THEN
                    OUTPUT "Enter new service name: "
                    INPUT new_name
                    OUTPUT "Enter new price: "
                    INPUT new_price
                    SET parts[1] TO new_name
                    SET parts[2] TO new_price
                    JOIN parts BY "|" INTO line
                    SET found TO True
                ENDIF
                APPEND line TO updated_list
            ENDFOR
            CLOSE file

            IF found EQUALS True THEN
                OPEN "services.txt" FOR WRITING
                FOR EACH item IN updated_list
                    WRITE item TO file
                ENDFOR
                CLOSE file
                OUTPUT "Service updated successfully"
            ELSE
                OUTPUT "Service not found"
            ENDIF

       
  ELSE IF service_update EQUALS 4 THEN
            OUTPUT "Enter service ID to delete: "
            INPUT target_id
            SET found TO False
            INITIALIZE remaining_services AS EMPTY LIST

            OPEN "services.txt" FOR READING
            FOR EACH line IN file
                SPLIT line BY "|" INTO parts
                IF parts[0] EQUALS target_id THEN
                    SET found TO True
                ELSE
                    APPEND line TO remaining_services
                ENDIF
            ENDFOR
            CLOSE file

            IF found EQUALS True THEN
                OPEN "services.txt" FOR WRITING
                FOR EACH item IN remaining_services
                    WRITE item TO file
                ENDFOR
                CLOSE file
                OUTPUT "Service has been deleted successfully"
            ELSE
                OUTPUT "Service not found"
            ENDIF

  ELSE IF service_update EQUALS 5 THEN
            BREAK WHILE LOOP

  ENDIF
    ENDWHILE
END service_menu
